"use client";

/**
 * MarginObjectLayer: the PING object plane beside the document plane.
 * (Nolan, 2026-09-22: "Margins Are the Object Layer".)
 *
 * DOCUMENT PLANE + OBJECT PLANE. Objects are never inserted into the
 * document; they live in fixed-position margin rails (an overlay layer),
 * so publisher content geometry is invariant to the layer's presence.
 *
 * - Anchoring: each object tracks a document anchor
 *   ([data-object-anchor="<key>"]); candidate Y = anchor midpoint.
 * - Placement: deterministic placeObjects() (see ./placement): clamp to
 *   viewport, resolve collisions with minimum vertical spacing, merge
 *   overflow into "+N" clusters. Stable input -> stable placement.
 * - Expansion: fixed overlay card in the margin layer. Never reflows.
 * - One-expanded: shared claim registry (./expansion) across circles,
 *   cluster popovers, and the mobile sheet.
 *
 * Responsive policy (functional, not polish):
 * - wide desktop (>=1400px): two object rails (left + right)
 * - normal desktop (1024-1399): one preferred rail + overflow clustering
 * - tablet (768-1023): compact edge rail
 * - mobile (<768): edge tab -> bottom sheet (no desktop margin emulation)
 *
 * Generic: descriptors only. Zero customer-specific code.
 */

import * as React from "react";
import { placeObjects, type AnchorInput, type Placed, type Rail } from "./placement";
import { ClusterSlot, EXPANDED_D, MarginObject, REST_D } from "./ObjectCircle";
import { MobileSheet } from "./MobileSheet";
import { claimExpanded } from "./expansion";
import type { MarginObjectDescriptor } from "./types";

export type { MarginObjectDescriptor };

const WIDE_MIN = 1400;
const DESKTOP_MIN = 1024;
const TABLET_MIN = 768;

type Projection = "dual" | "single" | "edge" | "mobile";

function projectionForWidth(w: number): Projection {
  if (w >= WIDE_MIN) return "dual";
  if (w >= DESKTOP_MIN) return "single";
  if (w >= TABLET_MIN) return "edge";
  return "mobile";
}

/** Compact rest diameter for the tablet edge rail. */
const EDGE_REST_D = 48;
/** Viewport-edge offset of the rail per projection. */
const RAIL_OFFSET: Record<Exclude<Projection, "mobile">, number> = {
  dual: 24,
  single: 12,
  edge: 8,
};

interface Geometry {
  vw: number;
  vh: number;
  scrollY: number;
  docH: number;
  anchors: Record<string, number | null>;
}

/**
 * Measures viewport, scroll, and anchor midpoints. Re-runs on scroll
 * (rAF-throttled), resize, font load, and once shortly after mount for
 * late layout shifts. Returns null until the first client measurement
 * (SSR renders the empty shell).
 */
function useGeometry(anchorKeys: string[]): Geometry | null {
  const keysKey = anchorKeys.join("|");
  const [geom, setGeom] = React.useState<Geometry | null>(null);

  React.useEffect(() => {
    let raf = 0;
    let disposed = false;
    const keys = keysKey.split("|").filter(Boolean);
    const measure = () => {
      if (disposed) return;
      const scrollY = window.scrollY;
      const anchors: Record<string, number | null> = {};
      for (const k of keys) {
        const el = document.querySelector(`[data-object-anchor="${CSS.escape(k)}"]`);
        if (!el) {
          anchors[k] = null;
          continue;
        }
        const r = el.getBoundingClientRect();
        anchors[k] = r.top + scrollY + r.height / 2;
      }
      setGeom({
        vw: window.innerWidth,
        vh: window.innerHeight,
        scrollY,
        docH: document.documentElement.scrollHeight,
        anchors,
      });
    };
    const schedule = () => {
      cancelAnimationFrame(raf);
      raf = requestAnimationFrame(measure);
    };
    measure();
    window.addEventListener("resize", schedule, { passive: true });
    window.addEventListener("scroll", schedule, { passive: true });
    const t = window.setTimeout(measure, 800);
    const fonts = (document as Document & { fonts?: { ready?: Promise<unknown> } }).fonts;
    if (fonts?.ready) {
      fonts.ready.then(() => measure()).catch(() => {});
    }
    return () => {
      disposed = true;
      cancelAnimationFrame(raf);
      window.removeEventListener("resize", schedule);
      window.removeEventListener("scroll", schedule);
      window.clearTimeout(t);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [keysKey]);

  return geom;
}

export function MarginObjectLayer({ objects }: { objects: MarginObjectDescriptor[] }) {
  const idsKey = objects.map((o) => o.objectId).join("|");
  // eslint-disable-next-line react-hooks/exhaustive-deps
  const anchorKeys = React.useMemo(() => objects.map((o) => o.anchorKey), [idsKey]);
  const geom = useGeometry(anchorKeys);
  const [sheetOpen, setSheetOpen] = React.useState(false);

  const projection: Projection | null = geom ? projectionForWidth(geom.vw) : null;

  const placed = React.useMemo<Placed[]>(() => {
    if (!geom || !projection || projection === "mobile") return [];
    const inputs: AnchorInput[] = objects.map((o) => ({
      objectId: o.objectId,
      anchorMidY: geom.anchors[o.anchorKey] ?? null,
      priority: o.priority,
    }));
    const restD = projection === "edge" ? EDGE_REST_D : REST_D;
    const rails: Rail[] = projection === "dual" ? ["right", "left"] : ["right"];
    return placeObjects(
      inputs,
      {
        viewportWidth: geom.vw,
        viewportHeight: geom.vh,
        scrollY: geom.scrollY,
        documentHeight: geom.docH,
      },
      {
        restDiameter: restD,
        minSpacing: restD + 16,
        edgePadding: 12,
        clusterWindow: 72,
        rails,
      },
    );
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [geom, projection, idsKey]);

  // eslint-disable-next-line react-hooks/exhaustive-deps
  const byId = React.useMemo(() => new Map(objects.map((o) => [o.objectId, o])), [idsKey]);

  /** Viewport-clamped fixed position for the 216px expanded card. */
  const cardStyleFor = React.useCallback(
    (cx: number, cy: number): React.CSSProperties => {
      if (!geom) return {};
      const half = EXPANDED_D / 2 + 8;
      const x = Math.min(Math.max(cx, half), Math.max(half, geom.vw - half));
      const y = Math.min(Math.max(cy, half), Math.max(half, geom.vh - half));
      return { left: x - EXPANDED_D / 2, top: y - EXPANDED_D / 2 };
    },
    [geom],
  );

  if (!geom || !projection) {
    // SSR / pre-measure shell: fixed overlay, zero document impact.
    return (
      <div
        data-testid="object-layer"
        aria-hidden="true"
        className="pointer-events-none fixed inset-0 z-50"
      />
    );
  }

  if (projection === "mobile") {
    return (
      <div
        data-testid="object-layer"
        data-projection="mobile"
        className="pointer-events-none fixed inset-0 z-50"
        style={{ overflowX: "clip" }}
      >
        {!sheetOpen && (
          <button
            type="button"
            data-testid="mobile-object-tab"
            onClick={() => {
              claimExpanded("sheet");
              setSheetOpen(true);
            }}
            aria-label={`${objects.length} PING objects on this page. Open object list.`}
            className="pointer-events-auto fixed right-0 top-1/2 flex min-h-[56px] w-[44px] flex-col items-center justify-center gap-1 rounded-l-xl text-white"
            style={{
              transform: "translateY(-50%)",
              background: "rgba(20,16,10,0.92)",
              boxShadow: "0 0 0 1px rgba(232,180,90,0.55)",
            }}
          >
            <span aria-hidden="true" className="flex flex-col gap-1">
              {objects.slice(0, 3).map((o) => (
                <span
                  key={o.objectId}
                  className="block h-2 w-2 rounded-full"
                  style={{ background: "#f3c96b" }}
                />
              ))}
            </span>
            <span className="text-[10px] font-bold">{objects.length}</span>
          </button>
        )}
        {sheetOpen && <MobileSheet objects={objects} onClose={() => setSheetOpen(false)} />}
      </div>
    );
  }

  const restD = projection === "edge" ? EDGE_REST_D : REST_D;
  const offset = RAIL_OFFSET[projection];
  const slotX = (rail: Rail) => (rail === "right" ? geom.vw - offset - restD : offset);

  return (
    <div
      data-testid="object-layer"
      data-projection={projection}
      className="pointer-events-none fixed inset-0 z-50"
      style={{ overflowX: "clip" }}
    >
      {placed.map((p) => {
        const x = slotX(p.rail);
        const cx = x + restD / 2;
        const slotStyle: React.CSSProperties = {
          left: x,
          top: p.y - restD / 2,
          width: restD,
          height: restD,
        };
        const cardStyle = cardStyleFor(cx, p.y);
        if (p.kind === "single") {
          const d = byId.get(p.objectId);
          if (!d) return null;
          return (
            <MarginObject
              key={p.objectId}
              descriptor={d}
              slotStyle={slotStyle}
              cardStyle={cardStyle}
              restD={restD}
            />
          );
        }
        const members = p.objectIds
          .map((id) => byId.get(id))
          .filter((m): m is MarginObjectDescriptor => !!m);
        if (members.length === 0) return null;
        // Cluster popover sits below the slot, clamped to the viewport.
        const popW = 240;
        const popLeft = Math.min(
          Math.max(cx - popW / 2, 8),
          Math.max(8, geom.vw - popW - 8),
        );
        const estPopH = members.length * 56 + 16;
        let popTop = p.y + restD / 2 + 8;
        if (popTop + estPopH > geom.vh - 8) {
          popTop = Math.max(8, p.y - restD / 2 - 8 - estPopH);
        }
        return (
          <ClusterSlot
            key={p.clusterId}
            clusterId={p.clusterId}
            members={members}
            slotStyle={slotStyle}
            popoverStyle={{ left: popLeft, top: popTop, width: popW }}
            cardStyleFor={() => cardStyle}
            restD={restD}
          />
        );
      })}
    </div>
  );
}

"use client";

/**
 * ObjectCircle: the generic PING object projection for the margin object
 * layer (Nolan, 2026-09-22: "Margins Are the Object Layer").
 *
 * ONE visual object, ONE interaction boundary:
 * - rest:     64px circle (48px on the tablet edge rail) with the object's
 *             image. Small, quiet, recognizable. No satellites.
 * - expanded: a 216px circle rendered as a FIXED overlay in the margin /
 *             overlay layer. It never reflows the publisher document:
 *             expansion is transient UI state, not document structure.
 *             Inside are the object name and exactly three actions
 *             (FOLLOW, LIKE, WEBSITE), all clipped inside the circular
 *             boundary. No control protrudes. No part of the circle is a
 *             secret link.
 *
 * Interaction:
 * - desktop: hover or keyboard focus expands; moving away collapses after
 *   a short forgiving delay (300ms); Escape collapses and returns focus
 *   to the trigger.
 * - touch (tablet edge rail): tap toggles; tap outside or the card
 *   background collapses. No hover assumption.
 * - keyboard: the rest trigger is a real button; focusing it opens the
 *   card and moves focus to the first enabled action; blur outside the
 *   object or Escape closes.
 *
 * Generic: takes object data as props. No per-customer logic lives here.
 * One-expanded coordination comes from ./expansion (module-level claim
 * registry shared by circles, cluster popovers, and the mobile sheet).
 */

import * as React from "react";
import { motion, useReducedMotion } from "framer-motion";
import { ArrowUpRight, ChevronLeft, Heart, UserCheck, UserPlus } from "lucide-react";
import { spring } from "@/motion/motionTokens";
import { claimExpanded, requestObjectOpen, subscribeExpanded, subscribeOpenRequest } from "./expansion";
import { useObjectActions } from "./use-object-actions";
import type { MarginObjectDescriptor } from "./types";

/** Resting diameter on desktop rails: compact, quiet, recognizable. */
export const REST_D = 64;
/** Expanded diameter: fits the three actions with 44px touch targets. */
export const EXPANDED_D = 216;
/** Forgiving collapse delay so the pointer can reach the controls. */
const COLLAPSE_DELAY_MS = 300;

const springTx = { type: "spring", ...spring.gentle } as const;
const instantTx = { duration: 0.01 } as const;

/**
 * Local open state for one claim id. Opening claims the global slot
 * (collapsing everyone else); being superseded collapses locally.
 * Closing is local-only: a close never releases another claimant's slot.
 */
export function useClaim(claimId: string, onSuperseded?: () => void) {
  const [open, setOpen] = React.useState(false);
  const timer = React.useRef<ReturnType<typeof setTimeout> | null>(null);

  const clearTimer = React.useCallback(() => {
    if (timer.current) {
      clearTimeout(timer.current);
      timer.current = null;
    }
  }, []);
  React.useEffect(() => clearTimer, [clearTimer]);

  const supersededRef = React.useRef(onSuperseded);
  supersededRef.current = onSuperseded;
  React.useEffect(
    () =>
      subscribeExpanded((active) => {
        if (active !== claimId) {
          clearTimer();
          setOpen(false);
          supersededRef.current?.();
        }
      }),
    [claimId, clearTimer],
  );

  const requestOpen = React.useCallback(() => {
    clearTimer();
    claimExpanded(claimId);
    setOpen(true);
  }, [claimId, clearTimer]);

  const requestClose = React.useCallback(() => {
    clearTimer();
    setOpen(false);
  }, [clearTimer]);

  const requestCloseSoon = React.useCallback(
    (focusInside?: () => boolean) => {
      clearTimer();
      timer.current = setTimeout(() => {
        // Never strand keyboard focus: if focus is inside, the blur
        // handler owns the close instead.
        if (!focusInside || !focusInside()) setOpen(false);
      }, COLLAPSE_DELAY_MS);
    },
    [clearTimer],
  );

  return { open, requestOpen, requestClose, requestCloseSoon, clearTimer };
}

interface CircleActionProps {
  label: string;
  stateLabel: string;
  pressed: boolean;
  disabled: boolean;
  onAction: () => void;
  buttonRef?: { current: HTMLButtonElement | null };
  children: React.ReactNode;
}

/**
 * One in-circle action. Visual footprint is compact (icon over a short
 * label); the hit target is always at least 44px tall.
 */
function CircleAction({
  label,
  stateLabel,
  pressed,
  disabled,
  onAction,
  buttonRef,
  children,
}: CircleActionProps) {
  return (
    <button
      ref={buttonRef}
      type="button"
      disabled={disabled}
      aria-pressed={pressed}
      aria-label={pressed ? stateLabel : label}
      title={pressed ? stateLabel : label}
      onClick={(e) => {
        e.stopPropagation();
        onAction();
      }}
      className="flex min-h-[44px] flex-1 flex-col items-center justify-center gap-0.5 rounded-xl px-1 text-white transition-colors disabled:opacity-50"
      style={{
        background: "rgba(0,0,0,0.62)",
        boxShadow: pressed
          ? "0 0 0 1px rgba(232,180,90,0.85), 0 2px 10px rgba(0,0,0,0.5)"
          : "0 0 0 1px rgba(255,255,255,0.30), 0 2px 10px rgba(0,0,0,0.5)",
        color: pressed ? "#f3c96b" : "#ffffff",
        backdropFilter: "blur(2px)",
      }}
    >
      {children}
      <span className="text-[10px] font-semibold leading-none">{label}</span>
    </button>
  );
}

/**
 * One margin object: rest button at the slot + fixed overlay card when
 * expanded. The card is a sibling of the trigger (never inside a
 * transformed element) so position:fixed stays viewport-relative.
 */
export function MarginObject({
  descriptor,
  slotStyle,
  cardStyle,
  restD = REST_D,
}: {
  descriptor: MarginObjectDescriptor;
  /** Fixed-position slot style from the placement layer. */
  slotStyle: React.CSSProperties;
  /** Fixed-position card style (viewport-clamped) from the layer. */
  cardStyle: React.CSSProperties;
  restD?: number;
}) {
  const { objectId, name } = descriptor;
  const reduceMotion = useReducedMotion();
  const tx = reduceMotion ? instantTx : springTx;
  const { open, requestOpen, requestClose, requestCloseSoon, clearTimer } = useClaim(objectId);

  // Peer jump: a related-object link in any card asks this slot to expand.
  React.useEffect(
    () =>
      subscribeOpenRequest((id) => {
        if (id === objectId) requestOpen();
      }),
    [objectId, requestOpen],
  );

  const wrapRef = React.useRef<HTMLDivElement>(null);
  const triggerRef = React.useRef<HTMLButtonElement | null>(null);
  const actionRefs: ActionRefs = {
    follow: React.useRef<HTMLButtonElement | null>(null),
    like: React.useRef<HTMLButtonElement | null>(null),
    web: React.useRef<HTMLButtonElement | null>(null),
  };
  const openedVia = React.useRef<"pointer" | "focus" | null>(null);
  // Escape returns focus to the trigger; the focus handler must not
  // treat that programmatic focus as a fresh keyboard open.
  const suppressFocusOpen = React.useRef(false);

  const focusInside = React.useCallback(
    () => !!wrapRef.current?.contains(document.activeElement),
    [],
  );

  // Keyboard open: move focus into the first enabled action so the
  // trigger (which hides while open) never strands focus. Re-runs when
  // the actions settle (they start disabled until the server confirms).
  const tryFocusFirstAction = React.useCallback(() => {
    if (openedVia.current !== "focus") return;
    openedVia.current = null;
    const first = [actionRefs.follow.current, actionRefs.like.current, actionRefs.web.current].find(
      (b) => b && !b.disabled,
    );
    first?.focus({ preventScroll: true });
  }, []);

  React.useEffect(() => {
    if (open) tryFocusFirstAction();
  }, [open, tryFocusFirstAction]);

  // While open: Escape collapses (focus returns to the trigger);
  // a pointerdown outside the object collapses. Deferred so the
  // opening tap never instantly dismisses.
  React.useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape") {
        suppressFocusOpen.current = true;
        requestClose();
        triggerRef.current?.focus({ preventScroll: true });
      }
    };
    const onDown = (e: PointerEvent) => {
      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) {
        requestClose();
      }
    };
    const t = setTimeout(() => document.addEventListener("pointerdown", onDown), 80);
    window.addEventListener("keydown", onKey);
    return () => {
      clearTimeout(t);
      window.removeEventListener("keydown", onKey);
      document.removeEventListener("pointerdown", onDown);
    };
  }, [open, requestClose]);

  const handleFocus = () => {
    if (suppressFocusOpen.current) {
      suppressFocusOpen.current = false;
      return;
    }
    openedVia.current = "focus";
    requestOpen();
  };

  const handleBlur = (e: React.FocusEvent) => {
    if (!wrapRef.current?.contains(e.relatedTarget as Node | null)) {
      requestCloseSoon(focusInside);
    }
  };

  const handleCardBackground = (e: React.MouseEvent) => {
    if ((e.target as HTMLElement).closest("button")) return;
    requestClose();
  };

  return (
    <div
      ref={wrapRef}
      data-testid="margin-object"
      data-object-id={objectId}
      className="pointer-events-auto"
      style={{ ...slotStyle, position: "fixed" }}
      onMouseEnter={() => {
        openedVia.current = "pointer";
        requestOpen();
      }}
      onMouseLeave={() => requestCloseSoon(focusInside)}
      onFocus={clearTimer}

###FILE:src/fyd/ui/object-layer/placement.ts
/**
 * Deterministic margin-placement algorithm for the PING object layer.
 * (Nolan, 2026-09-22: "Margins Are the Object Layer".)
 *
 * Pure module: no React, no DOM, no AI. Stable input produces stable
 * output. The layer measures anchors and viewport, then calls
 * placeObjects().
 *
 * Algorithm:
 *   1. Sort inputs by (priority, anchor Y, objectId). Total order, so
 *      placement is stable under input reordering.
 *   2. Cluster pass: anchors within clusterWindow of the cluster's first
 *      anchor merge into one slot (compact "+N" clustering).
 *   3. For each cluster: candidateY = anchor midpoint (document space)
 *      converted to viewport space, clamped to [lo, hi]. Try rails in
 *      preference order; resolve collisions by pushing down past occupied
 *      slots while keeping minimum vertical spacing. If pushed past the
 *      bottom edge, try the next rail.
 *   4. If no rail has room, merge into the nearest placed slot on the
 *      preferred rail (it becomes/extends a cluster). Objects never
 *      overlap; the worst case is a "+N" cluster, never a pile.
 */

export type Rail = "left" | "right";

export interface AnchorInput {
  objectId: string;
  /**
   * Document-space Y (px) of the anchor element's vertical midpoint,
   * or null when the anchor is missing. Missing anchors sort last and
   * clamp to the bottom edge: deterministic, and visibly wrong so a
   * missing anchor gets noticed instead of silently misplacing.
   */
  anchorMidY: number | null;
  /** Lower places first; ties break by anchor Y, then objectId. */
  priority: number;
}

export interface PlacementViewport {
  viewportWidth: number;
  viewportHeight: number;
  scrollY: number;
  documentHeight: number;
}

export interface PlacementOptions {
  /** Resting circle diameter (px). */
  restDiameter: number;
  /** Minimum center-to-center vertical spacing between slots in one rail. */
  minSpacing: number;
  /** Viewport inset (px) the slot center must respect, top and bottom. */
  edgePadding: number;
  /** Anchor-Y proximity (px) that merges objects into one cluster slot. */
  clusterWindow: number;
  /** Rails to use, in preference order. Empty = nothing placed. */
  rails: Rail[];
}

export interface PlacedSingle {
  kind: "single";
  objectId: string;
  rail: Rail;
  /** Viewport-space Y of the slot center (fixed-position coordinate). */
  y: number;
}

export interface PlacedCluster {
  kind: "cluster";
  /** Stable id derived from sorted member ids. */
  clusterId: string;
  objectIds: string[];
  rail: Rail;
  /** Viewport-space Y of the slot center (fixed-position coordinate). */
  y: number;
}

export type Placed = PlacedSingle | PlacedCluster;

function clamp(v: number, lo: number, hi: number): number {
  return Math.min(hi, Math.max(lo, v));
}

export function clusterIdFor(ids: string[]): string {
  return "cluster:" + [...ids].sort().join("+");
}

export function placeObjects(
  inputs: AnchorInput[],
  viewport: PlacementViewport,
  options: PlacementOptions,
): Placed[] {
  const { restDiameter, minSpacing, edgePadding, clusterWindow, rails } = options;
  if (rails.length === 0 || inputs.length === 0) return [];

  const lo = edgePadding + restDiameter / 2;
  const hi = Math.max(lo, viewport.viewportHeight - edgePadding - restDiameter / 2);
  const docH = Math.max(1, viewport.documentHeight);
  const anchorY = (i: AnchorInput): number => i.anchorMidY ?? docH;

  // 1. Total order: priority, then anchor Y, then objectId.
  const sorted = [...inputs].sort(
    (a, b) =>
      a.priority - b.priority ||
      anchorY(a) - anchorY(b) ||
      (a.objectId < b.objectId ? -1 : a.objectId > b.objectId ? 1 : 0),
  );

  // 2. Cluster pass: anchors within clusterWindow of the cluster's first
  //    anchor merge into one slot.
  const clusters: AnchorInput[][] = [];
  for (const input of sorted) {
    const last = clusters[clusters.length - 1];
    if (last && anchorY(input) - anchorY(last[0]) <= clusterWindow) {
      last.push(input);
    } else {
      clusters.push([input]);
    }
  }

  // 3. Place each cluster.
  const placed: Placed[] = [];
  const occupied = new Map<Rail, number[]>();
  for (const r of rails) occupied.set(r, []);

  for (const members of clusters) {
    const mean = members.reduce((s, m) => s + anchorY(m), 0) / members.length;
    const startY = clamp(mean - viewport.scrollY, lo, hi);
    let done = false;

    for (const rail of rails) {
      let cy = startY;
      const occ = occupied.get(rail)!; // kept ascending
      for (const oy of occ) {
        if (Math.abs(cy - oy) < minSpacing) cy = oy + minSpacing;
      }
      if (cy <= hi) {
        const idx = occ.findIndex((oy) => oy > cy);
        if (idx === -1) occ.push(cy);
        else occ.splice(idx, 0, cy);
        if (members.length === 1) {
          placed.push({ kind: "single", objectId: members[0].objectId, rail, y: cy });
        } else {
          const ids = members.map((m) => m.objectId);
          placed.push({
            kind: "cluster",
            clusterId: clusterIdFor(ids),
            objectIds: ids,
            rail,
            y: cy,
          });
        }
        done = true;
        break;
      }
    }

    // 4. No rail had room: merge into the nearest placed slot on the
    //    preferred rail. Never overlap; worst case is a "+N" cluster.
    if (!done) {
      const pref = rails[0];
      let best: Placed | null = null;
      let bestD = Infinity;
      for (const p of placed) {
        if (p.rail !== pref) continue;
        const d = Math.abs(p.y - startY);
        if (d < bestD) {
          bestD = d;
          best = p;
        }
      }
      const memberIds = members.map((m) => m.objectId);
      if (best && best.kind === "single") {
        const ids = [...new Set([best.objectId, ...memberIds])].sort();
        placed[placed.indexOf(best)] = {
          kind: "cluster",
          clusterId: clusterIdFor(ids),
          objectIds: ids,
          rail: best.rail,
          y: best.y,
        };
      } else if (best && best.kind === "cluster") {
        const ids = [...new Set([...best.objectIds, ...memberIds])].sort();
        best.objectIds = ids;
        best.clusterId = clusterIdFor(ids);
      } else {
        // Pathological (no placed slots at all): single cluster at startY.
        const ids = [...memberIds].sort();
        const occ = occupied.get(pref)!;
        occ.push(startY);
        occ.sort((a, b) => a - b);
        placed.push({
          kind: "cluster",
          clusterId: clusterIdFor(ids),
          objectIds: ids,
          rail: pref,
          y: startY,
        });
      }
    }
  }

  return placed;
}

 * circle, the cluster member card, and the mobile sheet card.
 *
 * The UI only ever reflects server-confirmed state (null = unknown, the
 * actions stay disabled until the server confirms).
 *
 * CANONICAL WRITE SEAM NOTE: executeFollow / executeLike POST to
 * /api/fyd/follow and /api/fyd/like, which write to the server-side
 * stores in src/fyd/object/follows.ts and src/fyd/object/likes.ts
 * (atomic JSON files, single local demo viewer until real identity
 * exists). The PING gateway envelope path in src/fyd/social/actions.ts
 * (signed RELATIONSHIP_CREATED) is the canonical social write seam but is
 * NOT wired into these routes; wiring it in requires a real viewer

 * toggles as gateway-journaled follows.

import { executeFollow, executeLike, openWebsite } from "@/fyd/capabilities/runtime";

export interface ObjectActions {
  following: boolean | null;
  liked: boolean | null;
  toggleFollow: () => Promise<void>;
  toggleLike: () => Promise<void>;
  openSite: () => void;
}

export function useObjectActions(objectId: string, websiteUrl: string | null): ObjectActions {
  const [following, setFollowing] = React.useState<boolean | null>(null);
  const [liked, setLiked] = React.useState<boolean | null>(null);

  React.useEffect(() => {
    let cancelled = false;
    setFollowing(null);
    setLiked(null);
    fetch(`/api/fyd/follow?objectId=${encodeURIComponent(objectId)}`)
      .then((r) => r.json())
      .then((d) => {
        if (!cancelled && d && d.ok === true) setFollowing(!!d.following);
      })
      .catch(() => {});
    fetch(`/api/fyd/like?objectId=${encodeURIComponent(objectId)}`)
      .then((r) => r.json())
      .then((d) => {
        if (!cancelled && d && d.ok === true) setLiked(!!d.liked);
      })
      .catch(() => {});
    return () => {
      cancelled = true;
    };
  }, [objectId]);

  const toggleFollow = React.useCallback(async () => {
    if (following === null) return;
    setFollowing(await executeFollow(objectId, following));
  }, [objectId, following]);

  const toggleLike = React.useCallback(async () => {
    if (liked === null) return;
    setLiked(await executeLike(objectId, liked));
  }, [objectId, liked]);

  const openSite = React.useCallback(() => {
    if (websiteUrl) openWebsite(websiteUrl);
  }, [websiteUrl]);

  return { following, liked, toggleFollow, toggleLike, openSite };
}

          <p className="text-sm font-bold uppercase tracking-widest text-honey">PING objects</p>
          <button
            type="button"
            onClick={onClose}
            aria-label="Close object list"
            className="flex min-h-[44px] min-w-[44px] items-center justify-center rounded-full text-white"
            style={{ background: "rgba(255,255,255,0.08)" }}
          >
            <X className="h-5 w-5" aria-hidden="true" />
          </button>
        </div>

        {selected ? (
          <div>
            <div className="mb-3">
              <BackButton onBack={() => setSelectedId(null)} label="Back to object list" />
            </div>
            <ObjectCardStatic descriptor={selected} />
            <p className="mt-3 text-center text-sm font-semibold text-white/80">{selected.name}</p>
          </div>
        ) : (
          <ul className="flex flex-col gap-1">
            {objects.map((o) => (
              <li key={o.objectId}>
                <button
                  type="button"
                  onClick={() => setSelectedId(o.objectId)}
                  aria-label={`View ${o.name}`}
                  className="flex min-h-[56px] w-full items-center gap-3 rounded-xl p-2 text-left text-white hover:bg-white/10"
                >
                  <span className="relative block h-11 w-11 shrink-0 overflow-hidden rounded-full bg-neutral-800">
                    {o.imageSrc ? (
                      <img
                        src={o.imageSrc}
                        srcSet={o.imageSrcSet ?? undefined}
                        sizes="88px"
                        alt=""
                        aria-hidden="true"
                        decoding="async"
                        className="absolute inset-0 h-full w-full object-cover"
                      />
                    ) : (
                      <span
                        aria-hidden="true"
                        className="absolute inset-0 flex items-center justify-center text-lg font-bold text-white"
                        style={{
                          background:
                            "radial-gradient(circle at 35% 30%, #8a6f4d, #3a3226)",
                        }}
                      >
                        {o.name.trim().charAt(0).toUpperCase() || "?"}
                      </span>
                    )}
                  </span>
                  <span className="line-clamp-2 flex-1 text-sm font-semibold leading-tight">
                    {o.name}
                  </span>
                  <ChevronRight className="h-5 w-5 shrink-0 text-white/50" aria-hidden="true" />
                </button>
              </li>
            ))}
          </ul>
        )}
      </div>
    </div>
  );
}
