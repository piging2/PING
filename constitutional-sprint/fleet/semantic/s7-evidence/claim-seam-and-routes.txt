    73	export interface OperatorAttestationInput {
    74	  resourceId: string;
    75	  operatorLabel: string;
    76	  basis: string;
    77	}

    79	/**
    80	 * Build an operator-attestation proof. Fails closed on a missing basis.
    81	 * The proof carries its limitation in proofNote so no surface can present
    82	 * it as stronger evidence than it is.
    83	 */
    84	export function attestOperatorControl(input: OperatorAttestationInput): VerificationProof {
    85	  const basis = (input.basis ?? "").trim();
    86	  if (basis.length < 10) {
    87	    throw new ClaimError(
    88	      "Operator attestation needs a written basis (at least 10 characters).",
    89	      "weak-proof",
    90	    );
    91	  }
    92	  if (!input.operatorLabel || !input.operatorLabel.trim()) {
    93	    throw new ClaimError("Operator attestation needs a named operator.", "weak-proof");
    94	  }
    95	  return {
    96	    method: "operator-attestation",
    97	    verifiedAt: new Date().toISOString(),
    98	    verifiedBy: input.operatorLabel.trim(),
    99	    basis,
   100	    artifacts: {
   101	      resourceId: input.resourceId,
   102	      attestation: "operator-statement",
   103	    },
   104	    proofNote:
   105	      "Operator attestation: a human operator asserts control. This is a statement, not a cryptographic proof of domain control.",
   106	  };
   107	}

     1	/**
     2	 * GET /api/fyd/claims/[resourceId]
     3	 *
     4	 * Read a resource claim. 404 when no record exists (missing data fails
     5	 * honest; nothing is invented). 400 on a malformed resource id.
     6	 */
     7	import { NextRequest, NextResponse } from "next/server";
     8	import { isValidResourceId } from "@/fyd/claim/machine";
     9	import { readClaim } from "@/fyd/claim/store";
    10	
    11	export const runtime = "nodejs";
    12	export const dynamic = "force-dynamic";
    13	
    14	export async function GET(
    15	  _req: NextRequest,
    16	  ctx: { params: Promise<{ resourceId: string }> },
    17	): Promise<NextResponse> {
    18	  const { resourceId } = await ctx.params;
    19	  if (!isValidResourceId(resourceId)) {
    20	    return NextResponse.json(
    21	      { ok: false, code: "invalid-resource-id", message: "Malformed resource id." },
    22	      { status: 400 },
    23	    );
    24	  }
    25	  const claim = readClaim(resourceId);
    26	  if (!claim) {
    27	    return NextResponse.json(
    28	      { ok: false, code: "not-found", message: "No claim record for this resource." },
    29	      { status: 404 },
    30	    );
    31	  }
    32	  return NextResponse.json({ ok: true, claim });
    33	}
