    72	function categoryFor(keywords: string | null): string | null {
    73	  if (!keywords) return null;
    74	  const first = keywords.split(",")[0]?.trim().toLowerCase() ?? "";
    75	  if (/carpent/.test(first)) return "Carpentry";
    76	  if (/plumb/.test(first)) return "Plumbing";
    77	  if (/hvac|heat|cool/.test(first)) return "HVAC";
    78	  if (!first) return null;

   100	/**
   101	 * Structured services on record: public service-role objects linked from
   102	 * the business by an active provides/offers relationship. Same selection
   103	 * the site generator's group() applies, so pages and ObjectView agree.
   104	 */
   105	function structuredServices(graph: ObjectGraph, businessId: string): PingObject[] {
   106	  const byId = new Map(graph.objects.map((o) => [o.id, o]));
   107	  const out = new Map<string, PingObject>();
   108	  for (const r of graph.relationships) {
   109	    if (r.subject !== businessId) continue;
   110	    if (r.status !== "active") continue;
   111	    if (!SERVICE_PREDICATES.includes(r.predicate)) continue;
   112	    const target = byId.get(r.object);
   113	    if (
   114	      target &&
   115	      SCHEMA_ROLES.service.includes(target.schema) &&
   116	      target.visibility === "public"
   117	    ) {
   118	      out.set(target.id, target);
   119	    }
   120	  }
   121	  return [...out.values()].sort((a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
   122	}

   124	function buildServices(
   125	  structured: { id: string; name: string }[],
   126	  objectId: string,
   127	): { services: ObjectServiceView[]; knownIds: string[]; knownNames: Map<string, string> } {
   128	  const overrides = readOverrides(objectId);
   129	  const knownIds: string[] = [];
   130	  const knownNames = new Map<string, string>();
   131	  const basis: ObjectServiceView[] = structured.map(({ id, name }) => {
   132	    knownIds.push(id);
   133	    knownNames.set(id, name);
   134	    return { id, name, basis: "structured" as const, basisLabel: "From the site data", visible: true };
   135	  });
   136	  for (const added of overrides.addedServices) {
   137	    if (knownIds.includes(added.id)) continue;
   138	    knownIds.push(added.id);
   139	    knownNames.set(added.id, added.name);
   140	    basis.push({
   141	      id: added.id,
   142	      name: added.name,
   143	      basis: "owner",
   144	      basisLabel: "Added by the owner",
   145	      visible: !overrides.hiddenServices.includes(added.id),
   146	    });
   147	  }
   148	
   149	  // Owner order wins; any ids not yet ordered keep structured order at the end.
   150	  const ordered: ObjectServiceView[] = [];
   151	  const byId = new Map(basis.map((s) => [s.id, s]));
   152	  for (const id of overrides.serviceOrder) {
   153	    const s = byId.get(id);
   154	    if (s) {
   155	      ordered.push(s);
   156	      byId.delete(id);
   157	    }
   158	  }
   159	  for (const s of basis) if (byId.has(s.id)) ordered.push(s);
   160	
   161	  const hidden = new Set(overrides.hiddenServices);
   162	  return {
   163	    services: ordered.map((s) => ({ ...s, visible: !hidden.has(s.id) })),
   164	    knownIds,
   165	    knownNames,
   166	  };
   167	}

   169	export function buildCapabilities(
   170	  contact: ObjectContactView,
   171	  evidence: { summary: string; services: ObjectServiceView[] },
   172	): ObjectCapability[] {
   173	  // "view" is demoted: the full customer page is no longer a capability
   174	  // (the type variant stays for compatibility, but it is never emitted).
   175	  // "follow" is a PING relationship, always available for a business object.
   176	  // call/email/website appear only when the underlying value exists.
   177	  // "ask" is evidence-gated: the ask pipeline answers from the object's
   178	  // own record (summary, services, contact), so a circle offers ask only
   179	  // when the object carries something to answer from. No ask evidence ->
   180	  // no ask action, never a dead question box.
   181	  const caps: ObjectCapability[] = [{ kind: "follow" }, { kind: "like" }];
   182	  const hasAskEvidence =
   183	    evidence.summary.trim().length > 0 ||
   184	    evidence.services.some((s) => s.visible) ||
   185	    contact.phone !== null ||
   186	    contact.email !== null ||
   187	    contact.website !== null ||
   188	    contact.locality !== null;
   189	  if (hasAskEvidence) caps.unshift({ kind: "ask" });
   190	  if (contact.phone) caps.push({ kind: "call", href: "tel:" + contact.phone.replace(/\s/g, ""), label: "Call" });
   191	  if (contact.email) caps.push({ kind: "email", href: "mailto:" + contact.email, label: "Email" });
   192	  if (contact.website) caps.push({ kind: "website", href: contact.website, label: "Website" });
   193	  return caps;
   194	}

   196	/**
   197	 * Shared composition core: build the ObjectView for ONE object in a
   198	 * tenant graph.
   199	 *
   200	 * - siteId selects the tenant (media manifests, never cross-tenant).
   201	 * - obj is the object the view describes (any schema, any public object).
   202	 * - viewId is the public id carried on the view.
   203	 * - overridesKey keys the owner store.
   204	 *
   205	 * The slug-keyed business path passes the site slug for viewId and
   206	 * overridesKey (legacy behavior, unchanged). The per-object path
   207	 * (loadObjectViewById in ./by-id) passes the PING object id for both.
   208	 *
   209	 * Not a public loader on its own: use loadObjectView / loadObjectViewById,
   210	 * which enforce the tenant/object honesty gates (unknown tenant, unknown
   211	 * id, non-public object).
   212	 */
   213	export function composeObjectView(
   214	  graph: ObjectGraph,
   215	  siteId: string,
   216	  obj: PingObject,
   217	  viewId: string,
   218	  overridesKey: string,
   219	): ObjectView {
   220	  const overrides = readOverrides(overridesKey);
   221	  const description = field(obj, "description") ?? obj.description ?? "";
   222	  const website = field(obj, "website");
   223	  const phone = field(obj, "phone");
   224	  const email = field(obj, "email");
   225	  const locality = field(obj, "locality");
   226	  const keywords = field(obj, "keywords");
   227	
   228	  const { services } = buildServices(
   229	    structuredServices(graph, obj.id).map((s) => ({ id: s.id, name: s.title })),
   230	    overridesKey,
   231	  );
   232	
   233	  // Media via the semantic attachment: the manifest is joined to the
   234	  // graph (Business represented_by Media) and the selector walks those
   235	  // relationships from the object. One truth for Page, Circle, and
   236	  // ObjectView; reference-only assets can never surface here.
   237	  const media: ObjectMediaView[] = listObjectMedia(siteId, graph, obj.id).map(
   238	    (d) => ({
   239	      id: d.id,
   240	      role: d.role,
   241	      src: d.src,
   242	      alt: d.alt,
   243	      rightsSource: d.rightsSource,
   244	      rightsBasis: d.rightsBasis,
   245	      sourceUrl: d.sourceUrl,
   246	      digest: d.digest,
   247	      observedAt: d.observedAt,
   248	    }),
   249	  );
   250	
   251	  const contact: ObjectContactView = {
   252	    phone,
   253	    email,
   254	    website,
   255	    locality,
   256	    addressVisibility: overrides.addressVisibility,
   257	  };
   258	
   259	  const domain = domainOf(website);
   260	  const firstService = services.find((s) => s.visible)?.name ?? services[0]?.name;
   261	
   262	  // Owner-attested field corrections, composed onto the graph by the read
   263	  // seam (src/fyd/data/ping-object-source.ts -> owner-overlay.ts). The
   264	  // contact block already carries the EFFECTIVE (owner-winning) values;
   265	  // these records keep the SOURCE SAYS X / OWNER SAYS Y distinction and
   266	  // the correction's own provenance for honest display.
   267	  const fieldCorrections: FieldCorrectionView[] = (
   268	    obj.ownerFieldCorrections ?? []
   269	  ).map((c) => ({ ...c, sourceDrifted: c.sourceDrifted ?? false }));
   270	
   271	  return {
   272	    id: viewId,
   273	    schema: obj.schema,
   274	    name: obj.title,
   275	    category: categoryFor(keywords),
   276	    locationLabel: locality,
   277	    summary: description,
   278	    media,
   279	    services,
   280	    serviceArea: parseServiceArea(field(obj, "area_served")),
   281	    contact,
   282	    capabilities: buildCapabilities(contact, { summary: description, services }),
   283	    provenance: {
   284	      kind: obj.provenance?.kind ?? "unknown",
   285	      ref: obj.provenance?.ref ?? "",
   286	      derivedAt: obj.provenance?.derivedAt ?? obj.updatedAt ?? "",
   287	      label: domain ? "Information observed on " + domain : "Information from the business website",
   288	    },
   289	    ownerUpdatedAt: overrides.history.length > 0 ? overrides.updatedAt : null,
   290	    fieldCorrections,
   291	    sampleQuestions: [
   292	      firstService ? "What kind of " + firstService.toLowerCase() + " work do you do?" : "What services do you offer?",
   293	      "Where do you work?",
   294	      "How can I get an estimate?",
   295	    ],
   296	  };
   297	}

   344	/** First maxChars of the summary, cut back to the last word boundary. */
   345	export function trimTagline(summary: string, maxChars: number): string {
   346	  const s = summary.trim();
   347	  if (s.length <= maxChars) return s;
   348	  const cut = s.slice(0, maxChars);
   349	  const lastSpace = cut.lastIndexOf(" ");
   350	  return (lastSpace > 0 ? cut.slice(0, lastSpace) : cut).trim();
   351	}

    86	/**
    87	 * Extract the acting tenant id from a context, or fail closed.
    88	 * Throws TenantContextError(TENANT_CONTEXT_MISSING) when the context is
    89	 * absent, malformed, or carries an invalid tenant id. Every store
    90	 * read/write path must call this (directly or via withTenantContext /
    91	 * assertTenantKey) before touching storage.
    92	 */
    93	export function requireTenantContext(
    94	  ctx: TenantContext | null | undefined,
    95	): string {
    96	  if (ctx == null || typeof ctx !== "object") {
    97	    throw new TenantContextError(
    98	      "TENANT_CONTEXT_MISSING",
    99	      "Refusing to act: no tenant context was provided. " +
   100	        "Background and request paths must carry an explicit tenant context.",
   101	    );
   102	  }
   103	  if (!isValidTenantId(ctx.tenantId)) {
   104	    throw new TenantContextError(
   105	      "TENANT_CONTEXT_MISSING",
   106	      "Refusing to act: the tenant context carries no valid tenant id.",
   107	    );
   108	  }
   109	  return ctx.tenantId;
   110	}
   111	
   112	/**
   113	 * Refuse a cross-tenant reference. The key a caller wants to read or
   114	 * write MUST belong to the acting tenant; anything else throws
   115	 * TenantContextError(CROSS_TENANT_REFERENCE). Returns the tenant id on
   116	 * success so wrappers can proceed with the verified key.
   117	 *
   118	 * This is the REJECTED-never-empty rule: the caller gets an exception,
   119	 * not null, not [].
   120	 */
   121	export function assertTenantKey(
   122	  ctx: TenantContext | null | undefined,
   123	  key: string,
   124	): string {
   125	  const tenantId = requireTenantContext(ctx);
   126	  if (typeof key !== "string" || key.length === 0) {
   127	    throw new TenantContextError(
   128	      "CROSS_TENANT_REFERENCE",
   129	      `Refusing: tenant "${tenantId}" presented an empty store key.`,
   130	      { tenantId, attemptedKey: key },
   131	    );
   132	  }
   133	  if (key !== tenantId) {
   134	    throw new TenantContextError(
   135	      "CROSS_TENANT_REFERENCE",
   136	      `Refusing: tenant "${tenantId}" attempted to reference ` +
   137	        `a key belonging to another tenant.`,
   138	      { tenantId, attemptedKey: key },
   139	    );
   140	  }
   141	  return tenantId;
   142	}
   143	
   144	/**
   145	 * Resolve the tenant-scoped subdirectory of a store base dir:
   146	 * <baseDir>/<tenantId>. The tenant id is validated against
   147	 * TENANT_ID_PATTERN, which also kills path traversal (.. , /, absolute
   148	 * paths) at the boundary. Stores that keep one file per tenant MUST
   149	 * build paths through here, never by raw string interpolation.
   150	 */
   151	export function tenantScopedDir(baseDir: string, tenantId: string): string {
   152	  if (!isValidTenantId(tenantId)) {
   153	    throw new TenantContextError(
   154	      "TENANT_CONTEXT_MISSING",
   155	      "Refusing to resolve a tenant directory for an invalid tenant id.",
   156	      { attemptedKey: tenantId },
   157	    );
   158	  }
   159	  // join() with a validated slug cannot escape baseDir.
   160	  // eslint-disable-next-line @typescript-eslint/no-var-requires
   161	  const { join } = require("node:path") as typeof import("node:path");
   162	  return join(baseDir, tenantId);
   163	}
   164	
   165	/**
   166	 * Run fn with a verified tenant context. The context is checked FIRST;
   167	 * fn never executes when the context is missing or invalid, which is
   168	 * what makes background/worker paths fail closed instead of running
   169	 * unscoped. The verified tenant id is handed to fn so the inner path
   170	 * can key storage without re-validating.
   171	 */
   172	export function withTenantContext<R>(
   173	  ctx: TenantContext | null | undefined,
   174	  fn: (tenantId: string) => R,
   175	): R {
   176	  const tenantId = requireTenantContext(ctx);
   177	  return fn(tenantId);
   178	}
