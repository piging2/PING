     2	 * Owner patch loop helpers for POST /api/fyd/objects/[objectId]/overrides.
     3	 *
     4	 * The loop: typed owner request -> proposal -> preview (before/after +
     5	 * evidence/capability impact) -> bound approval -> apply -> revert.
     6	 *
     7	 * Digest binding (STALE PROPOSAL): the propose stage returns
     8	 * baseStateDigest = sha256 over the canonicalized CURRENT owner state
     9	 * projection (the event-log projection, stable across reads) and
    10	 * patchDigest = sha256 over the canonicalized typed command. The approve
    11	 * stage recomputes the state digest against the live log; a mismatch means
    12	 * the state moved since the proposal was drafted and the approval is
    13	 * refused with 409 stale_proposal. Nothing is written on refusal.
    14	 *
    15	 * The approval record binds tenant + actor + base digest + patch digest +
    16	 * approval metadata + result digest, and is returned with every approve
    17	 * response alongside the chain audit trail.
    18	 *
    19	 * G4: every mutation requires a server-side TenantContext (derived from
    20	 * the trusted route path: the object id IS the site id) and an
    21	 * ActorContext. The actor on this route is always the seeded demo actor:
    22	 * DEMO OWNER CONTEXT is labeled on every response, never presented as a
    23	 * verified owner identity.

   106	/** The service order the site shows now: owner order first, then derived ids. */
   107	function currentOrder(objectId: string, ids: string[]): string[] {
   108	  const known = new Set(ids);
   109	  const owner = readOverrides(objectId).serviceOrder.filter((id) => known.has(id));
   110	  const order = [...owner];
   111	  for (const id of ids) if (!order.includes(id)) order.push(id);
   112	  return order;
   113	}

   115	/** What the order would be after applying a move-service command. */
   116	function movedOrder(order: string[], id: string, to: "up" | "down" | "first" | "last"): string[] {
   117	  const next = order.filter((x) => x !== id);
   118	  if (to === "first") next.unshift(id);
   119	  else if (to === "last") next.push(id);
   120	  else {
   121	    const idx = order.indexOf(id);
   122	    const at = to === "up" ? Math.max(0, idx - 1) : Math.min(next.length, idx + 1);
   123	    next.splice(at, 0, id);
   124	  }
   125	  return next;
   126	}

   139	/**
   140	 * Build the before/after + impact preview for a typed command. Pure read:
   141	 * interprets the command against the CURRENT state and describes the
   142	 * transition the approval would authorize. Returns null when the command
   143	 * cannot be previewed (the approve stage still validates independently).
   144	 */
   145	export function buildPatchPreview(
   146	  objectId: string,
   147	  command: OwnerCommand,
   148	  serviceIds: string[],
   149	  serviceNames: Map<string, string>,
   150	  capabilityImpact: string,
   151	): PatchPreview | null {
   152	  const overrides = readOverrides(objectId);
   153	  switch (command.type) {
   154	    case "set-contact-field": {
   155	      const existing = overrides.fieldCorrections[command.field];
   156	      const before = existing ? existing.ownerValue : (sourceFieldValue(objectId, command.field) ?? "(not on record)");
   157	      return {
   158	        before: command.field + ": " + before,
   159	        after: command.field + ": " + command.value,
   160	        evidenceImpact:
   161	          "Appends one owner.corrected-fact event (owner attestation) to this object's log. " +
   162	          "Ask FYD will cite the new value as an 'Owner-set value' from 'Owner correction'; " +
   163	          "the site's current value (" + before + ") stays recorded as what the source says. " +
   164	          "The /build page shows the owner value with a correction note.",
   165	        capabilityImpact,
   166	      };
   167	    }
   168	    case "revert-contact-field": {
   169	      const existing = overrides.fieldCorrections[command.field];
   170	      if (!existing) return null;
   171	      const source = sourceFieldValue(objectId, command.field) ?? "(not on record)";
   172	      return {
   173	        before: command.field + ": " + existing.ownerValue + " (owner correction)",
   174	        after: command.field + ": " + source + " (site's value shown again)",
   175	        evidenceImpact:
   176	          "Appends one owner.restored-fact event. The correction event stays in the log; " +
   177	          "the site's value is shown again on /build and in Ask FYD answers.",
   178	        capabilityImpact,
   179	      };
   180	    }
   181	    case "move-service": {
   182	      const order = currentOrder(objectId, serviceIds);
   183	      if (!order.includes(command.id)) return null;
   184	      const after = movedOrder(order, command.id, command.to);
   185	      const nameOf = (id: string) => serviceName(serviceNames, id);
   186	      return {
   187	        before: "Services order: " + order.map(nameOf).join(" > "),
   188	        after: "Services order: " + after.map(nameOf).join(" > "),
   189	        evidenceImpact:
   190	          "Appends one owner.corrected-fact event for services:order (owner attestation). " +
   191	          "The ObjectView service list re-projects from the log. " +
   192	          "Note: the /build page planner currently consumes owner field corrections, not service order.",
   193	        capabilityImpact,
   194	      };
   195	    }
   196	    case "set-service-visibility": {
   197	      if (!serviceIds.includes(command.id)) return null;
   198	      const hidden = overrides.hiddenServices.includes(command.id);
   199	      return {
   200	        before: serviceName(serviceNames, command.id) + ": " + (hidden ? "hidden" : "visible"),
   201	        after: serviceName(serviceNames, command.id) + ": " + (command.visible ? "visible" : "hidden"),
   202	        evidenceImpact:
   203	          "Appends one owner.hid-fact / owner.corrected-fact event (owner attestation). " +
   204	          "The ObjectView re-projects from the log. " +
   205	          "Note: the /build page planner currently consumes owner field corrections, not service visibility.",
   206	        capabilityImpact,
   207	      };
   208	    }
   209	    case "add-service": {
   210	      const order = currentOrder(objectId, serviceIds);
   211	      return {
   212	        before: order.length + " services: " + order.map((id) => serviceName(serviceNames, id)).join(", "),
   213	        after: order.length + 1 + " services: " + [...order.map((id) => serviceName(serviceNames, id)), command.name].join(", "),
   214	        evidenceImpact:
   215	          "Appends one owner.added-fact event (owner attestation). " +
   216	          "The ObjectView re-projects from the log.",
   217	        capabilityImpact,
   218	      };
   219	    }
   220	    case "set-address-visibility": {
   221	      return {
   222	        before: "address visibility: " + overrides.addressVisibility,
   223	        after: "address visibility: " + command.visibility,
   224	        evidenceImpact:
   225	          "Appends one owner event (owner attestation) for contact:address visibility. " +
   226	          "The ObjectView re-projects from the log.",
   227	        capabilityImpact,
   228	      };
   229	    }
   230	    case "confirm-contact-field": {
   231	      const correction = overrides.fieldCorrections[command.field];
   232	      const effective = correction
   233	        ? correction.ownerValue
   234	        : sourceFieldValue(objectId, command.field);
   235	      if (effective === null) return null;
   236	      return {
   237	        before: command.field + ": " + effective,
   238	        after: command.field + ": " + effective + " (owner confirms correct)",
   239	        evidenceImpact:
   240	          "Appends one owner.confirmed-fact event (owner attestation) " +
   241	          "recording that the owner asserts the current " +
   242	          command.field +
   243	          " is correct, with actor and timestamp. Nothing is rewritten: the " +
   244	          "source record is unchanged, and the confirmation keeps the " +
   245	          "source's value at confirmation time so later source drift can be " +
   246	          "detected (sourceDrifted).",
   247	        capabilityImpact,
   248	      };
   249	    }
   250	    default:
   251	      return null;
   252	  }
   253	}

    18	 *      for the controller relationship under the demo mapping.
    19	 *   5. PROPOSE/APPLY RULE: enforced by the caller (the overrides route's
    20	 *      propose/approve two-step). The caller must check verdict.allowed
    21	 *      BEFORE applying; approve appends the event, and the read model
    22	 *      re-projects from the log (EVENT -> PROJECTION).
    23	 *
    24	 * THIS IS NOT AUTHENTICATION. See ./gate.ts. Nothing here may back a
    25	 * production authorization decision. Production owner auth must be built

   127	  deps: CallbackDeps,
   128	  params: { code: string | null; state: string | null; stateCookie: string | null | undefined }
   129	): Promise<CallbackResult> {
   130	  requireReady(deps.config);
   131	  const { google, session } = deps.config;
   132	  const fetchImpl = deps.fetchImpl ?? fetch;
   133	  const nowIso = deps.nowIso ?? new Date().toISOString();
   134	
   135	  const oauthState = readOAuthStateValue(params.stateCookie, session.secret);
   136	  if (!oauthState) throw new AuthFlowError("BAD_STATE", "OAuth state missing, expired, or tampered", 400);
   137	  if (!params.state || params.state !== oauthState.state) {
   138	    throw new AuthFlowError("STATE_MISMATCH", "OAuth state mismatch", 400);
   139	  }
   140	  if (!isAllowedRedirectUri(google, oauthState.redirectUri)) {
   141	    throw new AuthFlowError("REDIRECT_NOT_ALLOWED", "Redirect URI is not allowlisted", 400);
   142	  }
   143	  if (!params.code) throw new AuthFlowError("MISSING_CODE", "Authorization code missing", 400);
   144	
   145	  let tokens;
   146	  try {
   147	    tokens = await exchangeCodeForTokens(
   148	      google,
   149	      { code: params.code, codeVerifier: oauthState.codeVerifier, redirectUri: google.redirectUri },
   150	      fetchImpl
   151	    );
   152	  } catch (err) {
   153	    throw new AuthFlowError("EXCHANGE_FAILED", "Could not exchange the authorization code", 502);
   154	  }
   155	  let verified: VerifiedAccount;
   156	  try {
   157	    verified = await verifyGoogleIdToken(
   158	      google,
   159	      tokens.idToken,
   160	      { expectedNonce: oauthState.nonce },
   161	      fetchImpl
   162	    );
   163	  } catch (err) {
   164	    throw new AuthFlowError("ID_TOKEN_INVALID", "Google identity could not be verified", 401);
   165	  }
   166	  // Provider tokens are intentionally dropped here: they never enter
   167	  // binding records, audit logs, journal payloads, or generated pages.
   168	  tokens = null;
   169	
   170	  const linkingTo = oauthState.linkIdentityId;
   171	  let result: { identityId: string; created: boolean; emailCollision: boolean; linked: boolean };
   172	  if (linkingTo) {
   173	    try {
   174	      const link = linkProviderAccount(
   175	        deps.store,
   176	        linkingTo,
   177	        verified,
   178	        { proofOfAuthentication: verified },
   179	        nowIso
   180	      );
   181	      result = { identityId: link.identityId, created: false, emailCollision: false, linked: !link.alreadyLinked };
   182	    } catch (err) {
   183	      if (err instanceof IdentityBindingError && err.code === "ACCOUNT_BOUND_ELSEWHERE") {
   184	        throw new AuthFlowError("ACCOUNT_BOUND_ELSEWHERE", "This Google account is linked to a different identity", 409);
   185	      }
   186	      throw err;
   187	    }
   188	    appendAuthAudit("link", {
   189	      identityId: linkingTo,
   190	      provider: verified.provider,
   191	      providerAccountId: providerAccountId(verified.provider, verified.issuer, verified.subject),
   192	      at: nowIso,
   193	    });
   194	  } else {
   195	    const signIn: SignInResult = signInOrSignUp(deps.store, verified, nowIso);
   196	    result = { identityId: signIn.identityId, created: signIn.created, emailCollision: signIn.emailCollision, linked: false };
   197	    appendAuthAudit(signIn.created ? "sign_up" : "sign_in", {
   198	      identityId: signIn.identityId,
   199	      provider: verified.provider,
   200	      providerAccountId: providerAccountId(verified.provider, verified.issuer, verified.subject),
   201	      note: signIn.emailCollision ? "email collision: separate identity, no merge" : undefined,
   202	      at: nowIso,
   203	    });
   204	  }
   205	
   206	  return {
   207	    identityId: result.identityId,
   208	    created: result.created,
   209	    emailCollision: result.emailCollision,
   210	    linked: result.linked,
   211	    sessionCookie: serializeSessionCookie(createSessionValue(result.identityId, session.secret), session),
   212	    clearStateCookie: clearOAuthStateCookie(session),
   213	  };
   214	}
