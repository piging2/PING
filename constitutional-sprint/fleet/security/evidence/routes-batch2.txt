     2	 * FYD social: follow / unfollow.
     3	 * POST   { actorId, targetId } -> 201 { ok:true, deduped, actionId, relationshipId, status, eventId, sequence }

     5	 *
     6	 * Real PING path: signed bff-envelope@1 RELATIONSHIP_CREATED (predicate
     7	 * 'follows') submitted to the repo-booted gateway. Every success is
     8	 * attested: the journal is re-read and the named preconditions checked
     9	 * before ok:true is returned. A failed follow returns ok:false with the
    10	 * missing evidence named (HTTP 502) — never a false success.
    11	 */
    12	
    13	import { NextRequest, NextResponse } from "next/server";
    14	import { followIdentity, unfollowIdentity, EnvelopeRejectedError } from "@/fyd/social/actions";
    15	
    16	export const dynamic = "force-dynamic";
    17	export const runtime = "nodejs";
    18	
    19	export async function POST(request: NextRequest) {
    20	  try {
    21	    const body = (await request.json().catch(() => null)) as {
    22	      actorId?: unknown;
    23	      targetId?: unknown;
    24	    } | null;
    25	    const actorId = typeof body?.actorId === "string" ? body.actorId : "";
    26	    const targetId = typeof body?.targetId === "string" ? body.targetId : "";
    27	    if (!actorId || !targetId) {
    28	      return NextResponse.json(
    29	        { ok: false, error: "actorId and targetId are required." },
    30	        { status: 400 },
    31	      );
    32	    }
    33	    const result = await followIdentity(actorId, targetId);
    34	    if (!result.ok) {
    35	      return NextResponse.json(result, { status: 502 });
    36	    }
    37	    return NextResponse.json(result, { status: 201 });
    38	  } catch (err) {
    39	    const message = err instanceof Error ? err.message : "unknown error";
    40	    const status = err instanceof EnvelopeRejectedError ? 502 : 400;
    41	    return NextResponse.json({ ok: false, error: message }, { status });
    42	  }
    43	}

    10	
    11	/**
    12	 * GET /api/fyd/like?objectId=xxx
    13	 *

    15	 * The endpoint never exposes other viewers and never lists all likes.

    24	      { ok: false, error: "objectId is required and must be lowercase letters, digits, or dashes." },
    25	      { status: 400 },

    27	  }
    28	  try {
    29	    return NextResponse.json({ ok: true, liked: isLiked(objectId) });
    30	  } catch (e) {
    31	    if (e instanceof LikeError) {
    32	      return NextResponse.json({ ok: false, error: "Unknown object." }, { status: 404 });
    33	    }
    34	    throw e;
    35	  }
    36	}

     1	import { NextResponse } from "next/server";
     2	import { getMediaProvenance, listMediaProvenance } from "@/fyd/media/provenance";
     3	
     4	export const dynamic = "force-dynamic";
     5	export const runtime = "nodejs";
     6	
     7	/**
     8	 * GET /api/fyd/media?siteId=<slug>&digest=<sha256>
     9	 *
    10	 * Provenance query surface for the managed FYD media pipeline
    11	 * (fyd-media@2). Returns the full provenance record for one asset:
    12	 * original source URL, observed_at, content digest (sha256), media type,
    13	 * dimensions, rights/source classification, and the derived-asset
    14	 * relationship (each variant names the original digest it was produced
    15	 * from). This is the record Ask FYD and WHY THIS cite; it never serves
    16	 * bytes and never hotlinks.
    17	 *
    18	 * GET /api/fyd/media?siteId=<slug> lists the provenance summaries for
    19	 * every acquired asset of the site.
    20	 *
    21	 * Failure honesty: unknown site or digest is a 404, never a plausible
    22	 * stand-in. Unclear-reference-only assets are reference-only by pipeline
    23	 * construction and never appear here.
    24	 */
    25	export async function GET(request: Request): Promise<NextResponse> {
    26	  const params = new URL(request.url).searchParams;
    27	  const siteId = params.get("siteId")?.trim() ?? "";
    28	  const digest = params.get("digest")?.trim() ?? "";
    29	  if (!siteId) {
    30	    return NextResponse.json(
    31	      { ok: false, error: "siteId query parameter is required." },
    32	      { status: 400 },
    33	    );
    34	  }
    35	  if (digest) {
    36	    const record = getMediaProvenance(siteId, digest);
    37	    if (!record) {
    38	      return NextResponse.json(
    39	        {
    40	          ok: false,
    41	          error:
    42	            "No acquired media asset with that digest for this site. " +
    43	            "Unknown digests and reference-only assets are never served.",
    44	        },
    45	        { status: 404 },
    46	      );
    47	    }
    48	    return NextResponse.json({ ok: true, ...record });
    49	  }
    50	  const records = listMediaProvenance(siteId);
    51	  if (!records) {
    52	    return NextResponse.json(
    53	      { ok: false, error: "Unknown siteId: no media manifest." },
    54	      { status: 404 },
    55	    );
    56	  }
    57	  return NextResponse.json({ ok: true, siteId, assets: records });
    58	}
