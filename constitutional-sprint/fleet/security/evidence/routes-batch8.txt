    11	 *
    12	 * Body: { siteId: string, objectId?: string, question: string, mode?: "visitor" | "owner" }
    13	 *
    14	 * - mode "visitor" (default): anonymous, public objects only, no grants.
    15	 * - mode "owner": accepted for a future authenticated lane; until that lane
    16	 *   exists it is treated as visitor-safe and grants nothing.
    17	 *
    18	 * The subject tenant is constructed server-side into a TenantContext from
    19	 * the validated body siteId (G3 hardening: previously no tenant context
    20	 * existed on this route). New callers should prefer the trusted-path route
    21	 * POST /api/fyd/ask/[siteId], where the tenant comes from the route path
    22	 * and any body-supplied tenant identity is refused on mismatch.
    23	 *
    24	 * Response: { ok: true, answer, answerClass, refusal, citations,
    25	 *   unknowns, suggestedActions, proposal, tenantId }
    26	 *   unknowns is string[]; suggestedActions is the available-actions list
    27	 *   ([] when none); proposal is a draft AskProposal or null.
    28	 *   answerClass is "supported" | "derived" | "unknown" (unknown = refusal).
    29	 *   Every citation carries claimClass ("supported" | "derived").
    30	 * Errors: 404 unknown site, 400 bad question or mode, 400 tenant_mismatch /
    31	 * invalid_tenant, 503 projection unavailable (answer unknown), 500 bundle
    32	 * invalid or unexpected internal failure. Every failure is structured JSON
    33	 * naming the failure kind; the answer is unknown in every failure case
    34	 * (answerUnknown: true), never an empty body.
    35	 */
    36	export async function POST(request: NextRequest): Promise<NextResponse> {
    37	  return handleAskRequest(null, request);

     7	/**
     8	 * POST /api/fyd/ask/[siteId] (trusted-path route; G3).
     9	 *
    10	 * Visitor Ask FYD: evidence-bounded Q&A over a site's PUBLIC objects only.
    11	 * The tenant is derived from the route path param, never from the request
    12	 * body or URL query: a TenantContext is constructed server-side via
    13	 * requireTenantContext (which also enforces the DNS-safe slug pattern).
    14	 *
    15	 * Body: { question: string, objectId?: string, mode?: "visitor" | "owner" }
    16	 *   objectId selects the target object inside the route tenant public graph
    17	 *   (unknown ids fail closed as 404 unknown_object).
    18	 *
    19	 * Any body-supplied tenant identity (siteId / tenantId / tenant keys) that
    20	 * disagrees with the route tenant is REFUSED with 400 tenant_mismatch: the
    21	 * request is served for the route tenant or refused, never for the
    22	 * body-claimed tenant. A matching claim is harmless and ignored.
    23	 *
    24	 * - mode "visitor" (default): anonymous, public objects only, no grants.
    25	 * - mode "owner": accepted for a future authenticated lane; until that lane
    26	 *   exists it is treated as visitor-safe and grants nothing.
    27	 *
    28	 * Response: { ok: true, answer, answerClass, refusal, citations,
    29	 *   unknowns, suggestedActions, proposal, tenantId }
    30	 *   unknowns is string[]; suggestedActions is the available-actions list
    31	 *   ([] when none); proposal is a draft AskProposal or null.
    32	 *   answerClass is "supported" | "derived" | "unknown" (unknown = refusal).
    33	 *   Every citation carries claimClass ("supported" | "derived").
    34	 */
    35	export async function POST(
    36	  request: NextRequest,
    37	  { params }: { params: Promise<{ siteId: string }> },
    38	): Promise<NextResponse> {
    39	  const { siteId } = await params;
    40	  return handleAskRequest(siteId, request);
    41	}

    36	async function runSharpDiagnostics(requestId: string) {
    37	  const results: any = {
    38	    requestId,

    41	  };
    42	
    43	  // Stage 1: Module import
    44	  try {
    45	    const sharp = require('sharp');
    46	    results.stages.moduleImport = {
    47	      status: 'passed',
    48	    };
    49	    results.stages.version = {
    50	      status: 'passed',
    51	      version: sharp.versions ? 'loaded' : 'unknown',
    52	    };
    53	
    54	    // Stage 2: Format capabilities (using documented APIs)
    55	    try {
    56	      results.stages.formatCapabilities = {
    57	        status: 'passed',
    58	        formats: sharp.format ? 'available' : 'unavailable',

    62	        status: 'failed',

   136	export async function GET(request: Request) {
   137	  const requestId = crypto.randomUUID();
   138	
   139	  // Simple header-based authorization for diagnostic endpoint
   140	  const authHeader = request.headers.get('authorization');
   141	  const expectedAuth = `Bearer ${process.env.DIAGNOSTIC_SECRET || 'dev-diagnostic-secret'}`;
   142	  
   143	  if (authHeader !== expectedAuth) {
   144	    return NextResponse.json(
   145	      {
   146	        success: false,
   147	        error: 'UNAUTHORIZED',
   148	        requestId,
   149	      },
   150	      { status: 401 }
   151	    );
   152	  }
   153	
   154	  try {
   155	    const diagnostics = await runSharpDiagnostics(requestId);
   156	    return NextResponse.json({
   157	      success: true,
   158	      diagnostics,
   159	      requestId,
   160	    });
   161	  } catch (error) {
   162	    return NextResponse.json(
   163	      {
   164	        success: false,
   165	        error: 'DIAGNOSTIC_ERROR',
   166	        requestId,
   167	        details: error instanceof Error ? error.message : 'Unknown error',
   168	      },
   169	      { status: 500 }
   170	    );
   171	  }
   172	}
