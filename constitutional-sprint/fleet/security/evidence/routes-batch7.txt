########## src/app/api/ping/proposal/route.ts ##########
     1	/**
     2	 * BFF: human approval for Ask PING proposals. The digest is re-verified
     3	 * server-side; any modification after drafting fails. Owner-only updates
     4	 * are enforced before the governed signed envelope is submitted as the
     5	 * viewer identity. Agents propose; they cannot publish.
     6	 *
     7	 * site_patch proposals additionally carry a signed canonical envelope and a
     8	 * siteSpecDigest binding; the current SiteSpec digest is re-read and
     9	 * compared before approval so a stale draft cannot apply to a changed spec.
    10	 * POST { proposal: AskProposal } -> { eventId: string }.
    11	 */
    12	
    13	import { NextRequest, NextResponse } from "next/server";
    14	import { getPingObjectReader, BadRequestError } from "@/lib/ping/ping-object-reader";
    15	import { verifySignedSitePatchDraft } from "@/fyd/ask/site-patch";
    16	import { getSiteSpecProvider } from "@/fyd/ask/site-spec";
    17	import { readerErrorResponse } from "@/lib/ping/api-errors";
    18	import { getPracticeIdentityId } from "@/lib/ping/session";
    19	import type { AskProposal } from "@/lib/ping/types";
    20	
    21	export const dynamic = "force-dynamic";
    22	export const runtime = "nodejs";
    23	
    24	function isProposal(v: unknown): v is AskProposal {
    25	  if (typeof v !== "object" || v === null) return false;
    26	  const r = v as Record<string, unknown>;
    27	  const base =
    28	    typeof r.schema === "string" &&
    29	    typeof r.digest === "string" &&
    30	    typeof r.changes === "object" &&
    31	    r.changes !== null;
    32	  if (r.kind === "object_update" || r.kind === "object_create") return base;
    33	  if (r.kind === "site_patch") {
    34	    const sp = r.sitePatch as Record<string, unknown> | null;
    35	    return (
    36	      base &&
    37	      typeof r.targetObjectId === "string" &&
    38	      typeof sp === "object" &&
    39	      sp !== null &&
    40	      typeof sp.siteSpecDigest === "string" &&
    41	      Array.isArray(sp.operations)
    42	    );
    43	  }
    44	  return false;
    45	}
    46	
    47	export async function POST(request: NextRequest) {
    48	  try {
    49	    const viewerId = await getPracticeIdentityId();
    50	    if (!viewerId) throw new BadRequestError("No active identity. Select an identity first.");
    51	    const body: unknown = await request.json().catch(() => null);
    52	    const rec = (typeof body === "object" && body !== null ? body : {}) as Record<string, unknown>;
    53	    if (!isProposal(rec.proposal)) throw new BadRequestError("proposal is required and must be a drafted AskProposal.");
    54	    const proposal = rec.proposal as AskProposal;
    55	    if (proposal.kind === "site_patch") {
    56	      // The FYD lane owns the signing key material: verify the draft's digest,
    57	      // its canonical envelope signature, and the envelope's digest binding.
    58	      if (!verifySignedSitePatchDraft(proposal)) {
    59	        throw new BadRequestError("Site-patch draft failed verification. Draft it again.");
    60	      }
    61	      // Re-read the CURRENT SiteSpec and compare digests: a draft reasoned
    62	      // over an older spec cannot apply after the spec changed.
    63	      const current = await getSiteSpecProvider()
    64	        .getSummary(proposal.sitePatch.siteId)
    65	        .catch(() => null);
    66	      if (!current) {
    67	        throw new BadRequestError("The site spec is unavailable; cannot verify the proposal is current.");
    68	      }
    69	      if (current.digest !== proposal.sitePatch.siteSpecDigest) {
    70	        throw new BadRequestError(
    71	          "The site changed since this proposal was drafted. Ask again for a fresh proposal.",
    72	        );
    73	      }
    74	    }
    75	    const result = await getPingObjectReader().submitProposal(viewerId, proposal);
    76	    return NextResponse.json(result);
    77	  } catch (err) {
    78	    return readerErrorResponse(err);
    79	  }
    80	}

     7	/**
     8	 * POST /api/fyd/ask (legacy flat route).
     9	 *
    10	 * Visitor Ask FYD: evidence-bounded Q&A over a site's PUBLIC objects only.
    11	 *
    12	 * Body: { siteId: string, objectId?: string, question: string, mode?: "visitor" | "owner" }
    13	 *
    14	 * - mode "visitor" (default): anonymous, public objects only, no grants.
    15	 * - mode "owner": accepted for a future authenticated lane; until that lane
    16	 *   exists it is treated as visitor-safe and grants nothing.
    17	 *
    18	 * The subject tenant is constructed server-side into a TenantContext from
    19	 * the validated body siteId (G3 hardening: previously no tenant context
    20	 * existed on this route). New callers should prefer the trusted-path route
    21	 * POST /api/fyd/ask/[siteId], where the tenant comes from the route path
    22	 * and any body-supplied tenant identity is refused on mismatch.
    23	 *
    24	 * Response: { ok: true, answer, answerClass, refusal, citations,
    25	 *   unknowns, suggestedActions, proposal, tenantId }
    26	 *   unknowns is string[]; suggestedActions is the available-actions list
    27	 *   ([] when none); proposal is a draft AskProposal or null.
    28	 *   answerClass is "supported" | "derived" | "unknown" (unknown = refusal).
    29	 *   Every citation carries claimClass ("supported" | "derived").
    30	 * Errors: 404 unknown site, 400 bad question or mode, 400 tenant_mismatch /
    31	 * invalid_tenant, 503 projection unavailable (answer unknown), 500 bundle
    32	 * invalid or unexpected internal failure. Every failure is structured JSON
    33	 * naming the failure kind; the answer is unknown in every failure case
    34	 * (answerUnknown: true), never an empty body.
    35	 */
    36	export async function POST(request: NextRequest): Promise<NextResponse> {
    37	  return handleAskRequest(null, request);
    38	}

    22	 * body-claimed tenant. A matching claim is harmless and ignored.
    23	 *
    24	 * - mode "visitor" (default): anonymous, public objects only, no grants.
    25	 * - mode "owner": accepted for a future authenticated lane; until that lane

    31	 *   ([] when none); proposal is a draft AskProposal or null.
    32	 *   answerClass is "supported" | "derived" | "unknown" (unknown = refusal).
    33	 *   Every citation carries claimClass ("supported" | "derived").
    34	 */
    35	export async function POST(
    36	  request: NextRequest,
    37	  { params }: { params: Promise<{ siteId: string }> },
