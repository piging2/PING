    69	async function handleParse(siteId: string, text: string) {
    70	  const parsed = parseCustomizationIntent(text);
    71	  if (isUnsupported(parsed)) {
    72	    return NextResponse.json(
    73	      { ok: false, code: "unsupported", error: parsed.reason, demo: true },
    74	      { status: 400 },
    75	    );
    76	  }
    77	  let site;
    78	  try {
    79	    site = await loadCustomizedSite(siteId);
    80	  } catch (e) {
    81	    return NextResponse.json(
    82	    demo: true,
    83	    demoOwnerMode: true,
    84	  });
    85	}

    87	export async function POST(req: Request) {
    88	  let body: ActionBody;
    89	  try {
    90	    body = (await req.json()) as ActionBody;
    91	  } catch {
    92	    return NextResponse.json({ ok: false, error: "Body must be JSON.", demo: true }, { status: 400 });
    93	  }
    94	  const action = body.action ?? "";
    95	  const siteId = (body.siteId ?? "").trim();
    96	  if (!siteId) {
    97	    return NextResponse.json({ ok: false, error: "siteId is required.", demo: true }, { status: 400 });
    98	  }
    99	  if (action === "parse") {
   100	    const text = (body.text ?? "").trim();
   101	    if (!text) {
   102	      return NextResponse.json({ ok: false, error: "text is required.", demo: true }, { status: 400 });
   103	    }
   104	    return handleParse(siteId, text);
   105	  }
   106	  if (action === "approve") {
   107	    const text = (body.text ?? "").trim();
   108	    if (!text) {
   109	      return NextResponse.json({ ok: false, error: "text is required.", demo: true }, { status: 400 });
   110	    }
   111	    if (!body.proposal) {
   112	      return NextResponse.json(
   113	        {
   114	          ok: false,
   115	          code: "no_proposal",
   116	          error:
   117	            "Approve requires the exact proposal the owner reviewed (from parse). " +
   118	            "Nothing is applied without an explicit, digest-verified approval.",
   119	          demo: true,
   120	        },
   121	        { status: 400 },
   122	      );
   123	    }
   124	    return handleApprove(req, siteId, text, body.proposal);
   125	  }
   126	  if (action === "clear") {
   127	    const intentId = (body.intentId ?? "").trim();
   128	    if (!intentId) {
   129	      return NextResponse.json({ ok: false, error: "intentId is required.", demo: true }, { status: 400 });
   130	    }
   131	    return handleClear(req, siteId, intentId);
   132	  }
   133	  return NextResponse.json(
   134	    { ok: false, error: "Unknown action. Use parse, approve, or clear.", demo: true },
   135	    { status: 400 },
   136	  );
   137	}

   109	/**
   110	 * Demo-owner-mode gate (same contract as the customize approval API):
   111	 * refuse with a typed 403 unless the demo explicitly opts in via
   112	 * NEXT_PUBLIC_FYD_DEMO_OWNER_MODE=1 AND the request arrived on a
   113	 * localhost/private-network host. Default off; absent/false => fail closed
   114	 * with { ok:false, code:"demo_owner_mode_required" }, before any I/O.

   116	function demoDenied(request: NextRequest) {
   117	  const enabled = isDemoOwnerModeEnabled();
   118	  const host = request.headers.get("host") ?? "";
   119	  const privateNet = isPrivateHost(host);
   120	  if (!enabled || !privateNet) {
   121	    return NextResponse.json(
   122	      {

   124	        code: "demo_owner_mode_required",

   126	          "Owner overrides require DEV/DEMO OWNER MODE " +
   127	          "(NEXT_PUBLIC_FYD_DEMO_OWNER_MODE=1) on a localhost or private-network host. " +
   128	          NOT_REAL_AUTH,
   129	        demoOwnerMode: enabled,
   130	        hostPrivate: privateNet,
   131	      },
   132	      { status: 403 },
   133	    );
   134	  }
   135	  return null;
